// ═══════════════════════════════════════════════════════════
// LOAD MODEL FILES INTO MEMFS
// ═══════════════════════════════════════════════════════════
async function loadModelFiles() {
statusEl.textContent = 'Downloading FR3 model…';

// Ensure directories exist
const dirs = new Set();
ALL_FILES.forEach(f => {
const parts = f.split('/');
let cur = '/working';
for (let i = 0; i < parts.length - 1; i++) {
cur += '/' + parts[i];
dirs.add(cur);
}
});
for (const d of dirs) {
if (!mujoco.FS.analyzePath(d).exists) {
mujoco.FS.mkdir(d);
}
}

let loaded = 0;
const total = ALL_FILES.length;

async function fetchFile(path) {
const url = MODEL_BASE + path;
const resp = await fetch(url);
if (!resp.ok) throw new Error(`Failed to fetch ${url}: ${resp.status}`);
const isBinary = path.endsWith('.stl') || path.endsWith('.png');
if (isBinary) {
const buf = await resp.arrayBuffer();
mujoco.FS.writeFile('/working/' + path, new Uint8Array(buf));
} else {
const text = await resp.text();
mujoco.FS.writeFile('/working/' + path, text);
}
loaded++;
statusEl.textContent = `Downloading FR3 model… (${loaded}/${total})`;
}

await Promise.all(ALL_FILES.map(fetchFile));
}

bodies[b].has_custom_mesh = false;
}

let geometry = new THREE.SphereGeometry(size[0] * 0.5);

if (type == 0) {
// Plane — skip (we have our own ground)
continue;
} else if (type == 2) {
geometry = new THREE.SphereGeometry(size[0]);
} else if (type == 3) {
geometry = new THREE.CapsuleGeometry(size[0], size[1] * 2.0, 20, 20);
} else if (type == 5) {
geometry = new THREE.CylinderGeometry(size[0], size[0], size[1] * 2.0);
} else if (type == 6) {
geometry = new THREE.BoxGeometry(size[0]*2, size[2]*2, size[1]*2);
} else if (type == 7) {
// Mesh geom
let meshID = mjModel.geom_dataid[g];
if (meshID < 0) continue;

if (!(meshID in meshGeometries)) {
geometry = new THREE.BufferGeometry();

// IMPORTANT: use .slice() to COPY data — subarray() is a view that
// would mutate MuJoCo's internal buffers when we swizzle.
let vertex_buffer = new Float32Array(mjModel.mesh_vert.subarray(
mjModel.mesh_vertadr[meshID] * 3,
(mjModel.mesh_vertadr[meshID] + mjModel.mesh_vertnum[meshID]) * 3));

// Swizzle Y/Z for three.js (on our copy, not the original)
for (let v = 0; v < vertex_buffer.length; v += 3) {
let temp = vertex_buffer[v + 1];
vertex_buffer[v + 1] = vertex_buffer[v + 2];
vertex_buffer[v + 2] = -temp;
}

let normal_buffer;
if (mjModel.mesh_normaladr && mjModel.mesh_normalnum) {
normal_buffer = new Float32Array(mjModel.mesh_normal.subarray(
mjModel.mesh_normaladr[meshID] * 3,
(mjModel.mesh_normaladr[meshID] + mjModel.mesh_normalnum[meshID]) * 3));
for (let v = 0; v < normal_buffer.length; v += 3) {
let temp = normal_buffer[v + 1];
normal_buffer[v + 1] = normal_buffer[v + 2];
normal_buffer[v + 2] = -temp;
}
}

let face_buffer = new Int32Array(mjModel.mesh_face.subarray(
mjModel.mesh_faceadr[meshID] * 3,
(mjModel.mesh_faceadr[meshID] + mjModel.mesh_facenum[meshID]) * 3));

geometry.setAttribute("position", new THREE.BufferAttribute(vertex_buffer, 3));
if (normal_buffer && normal_buffer.length === vertex_buffer.length) {
geometry.setAttribute("normal", new THREE.BufferAttribute(normal_buffer, 3));
}
geometry.setIndex(Array.from(face_buffer));
geometry.computeVertexNormals();

meshGeometries[meshID] = geometry;
} else {
// Clone so each geom instance gets its own transform
geometry = meshGeometries[meshID].clone();
}
bodies[b].has_custom_mesh = true;
} else {
continue;
}

// Material
let color;
if (mjModel.geom_matid[g] !== -1) {
let matId = mjModel.geom_matid[g];
color = [
mjModel.mat_rgba[(matId * 4) + 0],
mjModel.mat_rgba[(matId * 4) + 1],
mjModel.mat_rgba[(matId * 4) + 2],
mjModel.mat_rgba[(matId * 4) + 3]
];
} else {
color = [
mjModel.geom_rgba[(g * 4) + 0],
mjModel.geom_rgba[(g * 4) + 1],
mjModel.geom_rgba[(g * 4) + 2],
mjModel.geom_rgba[(g * 4) + 3]
];
}

let material = new THREE.MeshPhysicalMaterial({
color: new THREE.Color(color[0], color[1], color[2]),
transparent: color[3] < 1.0,
opacity: color[3],
roughness: 0.45,
metalness: 0.1,
clearcoat: 0.05,
});

let mesh = new THREE.Mesh(geometry, material);
mesh.castShadow = true;
mesh.receiveShadow = true;
mesh.bodyID = b;
bodies[b].add(mesh);
bodyMeshes.push(mesh);

// Set local geom pose within body
getPosition(mjModel.geom_pos, g, mesh.position);
getQuaternion(mjModel.geom_quat, g, mesh.quaternion);
}

// Add bodies to scene hierarchy
for (let b = 0; b < mjModel.nbody; b++) {
if (!(b in bodies)) {
bodies[b] = new THREE.Group();
bodies[b].bodyID = b;
bodies[b].has_custom_mesh = false;
}
if (b === 0) {
mujocoRoot.add(bodies[b]);
} else {
bodies[0].add(bodies[b]);
}
}

syncBodies();
}

// ═══════════════════════════════════════════════════════════
// COMPUTE DESIRED JOINT POSITIONS BASED ON MODE
// ═══════════════════════════════════════════════════════════
function updateDesiredPositions() {
for (let i = 0; i < NUM_JOINTS; i++) {
let offset = 0;
if (excitationMode === 'sine') {
// Sinusoidal excitation: 0.1 rad amplitude sine wave
offset = EXCITATION_AMPLITUDE * Math.sin(2 * Math.PI * SINE_FREQUENCY * simTime);
} else if (excitationMode === 'step') {
// Step excitation: toggle between +0.1 and -0.1 every STEP_PERIOD seconds
const phase = Math.floor(simTime / STEP_PERIOD) % 2;
offset = phase === 0 ? EXCITATION_AMPLITUDE : -EXCITATION_AMPLITUDE;
}
// else 'hold': offset stays 0
currentQDes[i] = HOME_QPOS[i] + offset;
}
}

// ═══════════════════════════════════════════════════════════
// PD CONTROL LOOP
// ═══════════════════════════════════════════════════════════
function applyPDControl() {
updateDesiredPositions();
for (let i = 0; i < NUM_JOINTS; i++) {
const kp = scalarKp * GAIN_BASE[i];
const kd = scalarKd * GAIN_BASE[i];
const posError = currentQDes[i] - mjData.qpos[i];
const velError = -mjData.qvel[i];
let torque = kp * posError + kd * velError;
torque = Math.max(-TORQUE_LIMITS[i], Math.min(TORQUE_LIMITS[i], torque));
mjData.ctrl[i] = torque;
}
}

}

// Recompute the world-space anchor from body-local point
function updateDragAnchor() {
if (!dragging) return;
const bodyGroup = bodies[dragBodyId];
dragPointWorld.copy(dragPointLocal);
bodyGroup.localToWorld(dragPointWorld);
}

// Compute and apply the spring force + update the arrow visual
function applyDragForce() {
if (!dragging) return;

const disp = dragCursorWorld.clone().sub(dragPointWorld);
const len = disp.length();

// Update arrow: starts at the body anchor, points toward cursor
if (len > 0.001) {
const arrowLen = Math.min(len, 0.5);
const dir = disp.clone().normalize();

// Position the group at the anchor point
dragArrow.position.copy(dragPointWorld);

// Orient: default cylinder points +Y, rotate to match dir
const quat = new THREE.Quaternion();
quat.setFromUnitVectors(new THREE.Vector3(0, 1, 0), dir);
dragArrow.quaternion.copy(quat);

// Scale shaft length, place head at tip
const shaft = dragArrow.getObjectByName('shaft');
const head = dragArrow.getObjectByName('head');
const headLen = Math.min(arrowLen * 0.35, 0.06);
const shaftLen = arrowLen - headLen;
shaft.scale.set(1, Math.max(shaftLen, 0.001), 1);
head.position.set(0, shaftLen, 0);
head.scale.set(1, 1, 1);

dragArrow.visible = true;
} else {
dragArrow.visible = false;
}

// Clear + apply perturbation
for (let i = 0; i < mjData.qfrc_applied.length; i++) {
mjData.qfrc_applied[i] = 0;
}

const mass = mjModel.body_mass[dragBodyId];
const forceScale = mass * 250;
const forceVec = toMujocoPos(disp.clone().multiplyScalar(forceScale));
const pointVec = toMujocoPos(dragPointWorld.clone());

mujoco.mj_applyFT(
mjModel, mjData,
new Float64Array([forceVec.x, forceVec.y, forceVec.z]),
new Float64Array([0, 0, 0]),
new Float64Array([pointVec.x, pointVec.y, pointVec.z]),
dragBodyId,
mjData.qfrc_applied
);
}

}
}

function animate(time) {

for (let i = 0; i < Math.min(nsteps, 30); i++) {

applyPDControl();

}

// Record & plot joint state (throttled to ~20 Hz)
plotFrameSkip++;
if (plotFrameSkip >= 3) {
plotFrameSkip = 0;

drawJointPlot();
}

syncBodies();
controls.update();
renderer.render(scene, camera);
}

// ═══════════════════════════════════════════════════════════
// KP/KD GRID CONTROL + BIDIRECTIONAL SYNC
// ═══════════════════════════════════════════════════════════
const KPS = [16, 32, 64, 128, 256, 512];
const KDS = [1, 2, 4, 8, 12, 16, 24];

// Normalized position on the grid [0,1] — log-interpolated
let gainNormX = 0.729; // → Kp≈200
let gainNormY = 0.725; // → Kd≈10
let gainDragging = false;
let gainGridPulseActive = false;
let gainGridPulsePhase = 0;

// Corner colors matching the PD response widget
const GC_A = [140,140,140], GC_B = [76,114,176], GC_C = [196,78,82], GC_D = [129,114,178];

function lerpColor(c1, c2, t) {
return [c1[0]+(c2[0]-c1[0])*t, c1[1]+(c2[1]-c1[1])*t, c1[2]+(c2[2]-c1[2])*t];
}
function gainColor(nx, ny) {
const bot = lerpColor(GC_A, GC_B, nx);
const top = lerpColor(GC_C, GC_D, nx);
const c = lerpColor(bot, top, ny);
return `rgb(${Math.round(c[0])},${Math.round(c[1])},${Math.round(c[2])})`;
}

function gainsFromNorm(nx, ny) {
const logKp = (1 - nx) * Math.log(KPS[0]) + nx * Math.log(KPS[KPS.length - 1]);
const logKd = (1 - ny) * Math.log(KDS[0]) + ny * Math.log(KDS[KDS.length - 1]);
return { kp: Math.exp(logKp), kd: Math.exp(logKd) };
}
function normFromGains(kp, kd) {
const nx = (Math.log(kp) - Math.log(KPS[0])) / (Math.log(KPS[KPS.length-1]) - Math.log(KPS[0]));
const ny = (Math.log(kd) - Math.log(KDS[0])) / (Math.log(KDS[KDS.length-1]) - Math.log(KDS[0]));
return { nx: Math.max(0, Math.min(1, nx)), ny: Math.max(0, Math.min(1, ny)) };
}

function applyGains(nx, ny, broadcast) {
gainNormX = Math.max(0, Math.min(1, nx));
gainNormY = Math.max(0, Math.min(1, ny));
const g = gainsFromNorm(gainNormX, gainNormY);
scalarKp = g.kp;
scalarKd = g.kd;
updateGainDisplay();
drawGainGrid();
updateSelectorColors(); // Update joint button color when gains change
if (broadcast) {
try {
window.parent.postMessage({ type: 'mujoco-gains-changed', kp: scalarKp, kd: scalarKd }, '*');
} catch(_) {}
}
}

function updateGainDisplay() {
document.getElementById('kp-val').textContent = Math.round(scalarKp);
document.getElementById('kd-val').textContent = Math.round(scalarKd * 10) / 10;
}

function startDemoPhysics() {
if (!mjModel || !mjData || !dragArrow) return;

// Pick a mid-arm body (skip world=0 and base=1)

updateDragAnchor();
dragCursorWorld.copy(dragPointWorld);

controls.enabled = false;
dragging = true;
demoDragActive = true;
demoDragStartTime = performance.now();

tickDemoDrag();
}

function tickDemoDrag() {
if (!demoDragActive) return;

const t = (performance.now() - demoDragStartTime) / 1000;

// tanh "squares" a slow sine so the force saturates at its peak and holds
// there for ~4–5 s before snapping briefly to the opposite direction.
// freq = 0.10 Hz → 10 s full period → ~5 s per push direction
// squeeze = 6 → flat tops, sharp zero-crossings
const amp = 0.25;
const freq = 0.20;
const squeeze = 6;
const phase = 2 * Math.PI * freq * t;

updateDragAnchor();

dragCursorWorld.copy(dragPointWorld);
dragCursorWorld.z += amp * Math.tanh(squeeze * Math.sin(phase));
dragCursorWorld.x += amp * 0.35 * Math.sin(2 * Math.PI * 0.07 * t); // slow independent X drift

requestAnimationFrame(tickDemoDrag);
}
